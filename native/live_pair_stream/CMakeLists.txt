cmake_minimum_required(VERSION 3.20)
project(ac6_live_pair_stream LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
option(AC6_FRAME_TRACE "Bounded projection/provenance diagnostics; never enables VR metadata" OFF)
option(AC6_RENDER_CONTEXT_TRACE "Observe CPU render context invocation scopes; not a GPU binding" OFF)
option(AC6_CONTEXT_FORWARD_TRACE "Read pinned CPU adapter forwarding candidates at entry; never dispatch or GPU proof" OFF)
if(AC6_CONTEXT_FORWARD_TRACE AND NOT AC6_RENDER_CONTEXT_TRACE)
  message(FATAL_ERROR "AC6_CONTEXT_FORWARD_TRACE requires AC6_RENDER_CONTEXT_TRACE")
endif()
if(AC6_RENDER_CONTEXT_TRACE AND NOT AC6_FRAME_TRACE)
  message(FATAL_ERROR "AC6_RENDER_CONTEXT_TRACE requires AC6_FRAME_TRACE")
endif()
add_executable(test_frame_trace test_frame_trace.cpp)
target_compile_definitions(test_frame_trace PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX AC6_FRAME_TRACE)
add_test(NAME stream_frame_provenance_rejects_mismatches COMMAND test_frame_trace)
add_executable(test_stream_protocol test_protocol.cpp)
target_compile_definitions(test_stream_protocol PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME stream_ring_ownership_and_validation COMMAND test_stream_protocol)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../reference/minhook minhook)
add_executable(test_projection_observer test_projection_observer.cpp)
target_compile_definitions(test_projection_observer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(test_projection_observer PRIVATE minhook)
add_test(NAME stream_projection_observer_is_unassociated_and_bounded COMMAND test_projection_observer)
add_executable(test_render_context test_render_context.cpp)
target_compile_definitions(test_render_context PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(test_render_context PRIVATE minhook)
add_test(NAME render_context_scope_is_bounded_and_thread_local COMMAND test_render_context)
add_executable(test_context_forwarding test_context_forwarding.cpp)
target_compile_definitions(test_context_forwarding PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(test_context_forwarding PRIVATE minhook)
add_test(NAME context_forwarding_is_guarded_and_immutable COMMAND test_context_forwarding)
set_tests_properties(context_forwarding_is_guarded_and_immutable PROPERTIES TIMEOUT 45)
add_executable(test_projection_context test_projection_context.cpp)
target_compile_definitions(test_projection_context PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX AC6_RENDER_CONTEXT_TRACE)
target_link_libraries(test_projection_context PRIVATE minhook)
add_test(NAME projection_helper_preserves_cpu_invocation_scope COMMAND test_projection_context)
set_tests_properties(render_context_scope_is_bounded_and_thread_local projection_helper_preserves_cpu_invocation_scope PROPERTIES TIMEOUT 45)
add_executable(test_live_pair_stream test_host.cpp probe.cpp ../frame_pair_probe/slots.c)
target_compile_definitions(test_live_pair_stream PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX LIVE_PAIR_TEST_HOST)
target_link_libraries(test_live_pair_stream PRIVATE minhook d3d12 dxgi user32)
add_test(NAME stream_pairs_remain_coherent_and_ring_drops_without_wait COMMAND test_live_pair_stream)
add_test(NAME stream_pending_gpu_and_stale_depth_gates COMMAND test_live_pair_stream --gates)
set_tests_properties(stream_pairs_remain_coherent_and_ring_drops_without_wait PROPERTIES TIMEOUT 45)
add_library(ac6_live_pair_probe SHARED probe.cpp ../frame_pair_probe/slots.c)
set(LIVE_PAIR_PROBE_NAME "ac6_live_pair_probe_v1" CACHE STRING "Unique unmapped DLL name")
set_target_properties(ac6_live_pair_probe PROPERTIES OUTPUT_NAME "${LIVE_PAIR_PROBE_NAME}")
target_compile_definitions(ac6_live_pair_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ac6_live_pair_probe PRIVATE minhook d3d12 dxgi user32)
if(AC6_FRAME_TRACE)
  target_compile_definitions(ac6_live_pair_probe PRIVATE AC6_FRAME_TRACE)
  target_compile_definitions(test_live_pair_stream PRIVATE AC6_FRAME_TRACE)
endif()
if(AC6_RENDER_CONTEXT_TRACE)
  target_compile_definitions(ac6_live_pair_probe PRIVATE AC6_RENDER_CONTEXT_TRACE)
  target_compile_definitions(test_live_pair_stream PRIVATE AC6_RENDER_CONTEXT_TRACE)
endif()
if(AC6_CONTEXT_FORWARD_TRACE)
  target_compile_definitions(ac6_live_pair_probe PRIVATE AC6_CONTEXT_FORWARD_TRACE)
  target_compile_definitions(test_live_pair_stream PRIVATE AC6_CONTEXT_FORWARD_TRACE)
endif()
