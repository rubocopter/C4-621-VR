cmake_minimum_required(VERSION 3.20)
project(ac6_live_pair_stream LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
add_executable(test_stream_protocol test_protocol.cpp)
target_compile_definitions(test_stream_protocol PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME stream_ring_ownership_and_validation COMMAND test_stream_protocol)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../reference/minhook minhook)
add_executable(test_live_pair_stream test_host.cpp probe.cpp ../frame_pair_probe/slots.c)
target_compile_definitions(test_live_pair_stream PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX LIVE_PAIR_TEST_HOST)
target_link_libraries(test_live_pair_stream PRIVATE minhook d3d12 dxgi user32)
add_test(NAME stream_pairs_remain_coherent_and_ring_drops_without_wait COMMAND test_live_pair_stream)
add_test(NAME stream_pending_gpu_and_stale_depth_gates COMMAND test_live_pair_stream --gates)
set_tests_properties(stream_pairs_remain_coherent_and_ring_drops_without_wait PROPERTIES TIMEOUT 45)
add_library(ac6_live_pair_probe SHARED probe.cpp ../frame_pair_probe/slots.c)
set(LIVE_PAIR_PROBE_NAME "ac6_live_pair_probe_v1" CACHE STRING "Unique unmapped DLL name")
set_target_properties(ac6_live_pair_probe PROPERTIES OUTPUT_NAME "${LIVE_PAIR_PROBE_NAME}")
target_compile_definitions(ac6_live_pair_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ac6_live_pair_probe PRIVATE minhook d3d12 dxgi user32)
