cmake_minimum_required(VERSION 3.20)
project(ac6_camera_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../reference/minhook minhook)
add_executable(test_yaw test_yaw.cpp)
enable_testing()
add_test(NAME yaw_preserves_position_and_rotates_local_basis COMMAND test_yaw)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/probe.cpp")
  add_library(ac6_camera_probe SHARED probe.cpp)
  target_link_libraries(ac6_camera_probe PRIVATE minhook)
  target_compile_definitions(ac6_camera_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_library(ac6_headtracking_probe SHARED probe.cpp)
  target_link_libraries(ac6_headtracking_probe PRIVATE minhook)
  target_compile_definitions(ac6_headtracking_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX AC6_HEADTRACKING_PROBE)
  add_library(ac6_headtracking_display_probe SHARED probe.cpp)
  target_link_libraries(ac6_headtracking_display_probe PRIVATE minhook)
  target_compile_definitions(ac6_headtracking_display_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX AC6_HEADTRACKING_PROBE AC6_HEADTRACKING_DISPLAY_PROBE)
  add_library(ac6_headtracking_calibrated_probe SHARED probe.cpp)
  set(HEADTRACKING_CALIBRATED_NAME "ac6_headtracking_calibrated_probe" CACHE STRING "Unique unmapped calibrated probe name")
  set_target_properties(ac6_headtracking_calibrated_probe PROPERTIES OUTPUT_NAME "${HEADTRACKING_CALIBRATED_NAME}")
  target_link_libraries(ac6_headtracking_calibrated_probe PRIVATE minhook)
  target_compile_definitions(ac6_headtracking_calibrated_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX AC6_HEADTRACKING_PROBE AC6_HEADTRACKING_CALIBRATED_PROBE)
endif()
