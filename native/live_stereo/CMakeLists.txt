cmake_minimum_required(VERSION 3.20)
project(ac6_live_stereo LANGUAGES CXX)
option(LIVE_STEREO_BUILD_TESTS "Build standalone hardware shared-GPU fixture" ON)
add_library(live_stereo STATIC live_stereo.cpp)
target_compile_features(live_stereo PUBLIC cxx_std_17)
target_include_directories(live_stereo PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(live_stereo PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(live_stereo PUBLIC d3d12 dxgi d3dcompiler)
if(MSVC)
  target_compile_options(live_stereo PRIVATE /W4 /EHsc)
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/reprojection.hlsl" "${CMAKE_CURRENT_SOURCE_DIR}/../reprojection/parallel_projection.hlsl")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../reprojection/parallel_projection.hlsl" PARALLEL_PROJECTION_HLSL)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/reprojection.hlsl" LIVE_STEREO_HLSL)
configure_file(shader.hpp.in shader.hpp @ONLY)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/monitor.cpp")
  add_executable(live_stereo_monitor monitor.cpp)
  target_link_libraries(live_stereo_monitor PRIVATE live_stereo)
endif()
if(LIVE_STEREO_BUILD_TESTS)
  enable_testing()
  add_executable(test_projection test_projection.cpp)
  target_compile_features(test_projection PRIVATE cxx_std_17)
  add_test(NAME projection_contract_validation COMMAND test_projection)
  add_executable(test_live_stereo test_live_stereo.cpp)
  target_include_directories(test_live_stereo PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(test_live_stereo PRIVATE live_stereo)
  add_test(NAME live_stereo_shared_hardware COMMAND test_live_stereo)
  add_test(NAME live_stereo_inflight_timeout COMMAND test_live_stereo --timeout)
  add_test(NAME live_stereo_output_1080p COMMAND test_live_stereo --1080p)
  add_test(NAME live_stereo_output_1440p COMMAND test_live_stereo --1440p)
  add_test(NAME live_stereo_rejects_output_resolution COMMAND test_live_stereo --invalid-resolution)
  add_test(NAME live_stereo_projection_models COMMAND test_live_stereo --projection-contract)
  add_test(NAME live_stereo_distinct_eye_intrinsics COMMAND test_live_stereo --eye-projections)
  add_test(NAME projection_rejects_nonfinite_gpu_y COMMAND test_live_stereo --nonfinite-projection)
  add_test(NAME projection_shader_mode_isolation COMMAND test_live_stereo --shader-isolation)
  set_tests_properties(projection_rejects_nonfinite_gpu_y PROPERTIES TIMEOUT 40)
  set_tests_properties(live_stereo_distinct_eye_intrinsics PROPERTIES TIMEOUT 120)
  set_tests_properties(live_stereo_projection_models PROPERTIES TIMEOUT 120)
  set_tests_properties(live_stereo_shared_hardware live_stereo_output_1080p live_stereo_output_1440p PROPERTIES TIMEOUT 120)
  set_tests_properties(live_stereo_inflight_timeout PROPERTIES TIMEOUT 40)
endif()
