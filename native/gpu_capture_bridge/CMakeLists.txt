cmake_minimum_required(VERSION 3.24)
project(gpu_capture_bridge LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows is required")
endif()
add_library(gpu_capture_bridge STATIC gpu_window_bridge.cpp gpu_image_transport.cpp)
target_compile_features(gpu_capture_bridge PUBLIC cxx_std_20)
target_compile_definitions(gpu_capture_bridge PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(gpu_capture_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(gpu_capture_bridge PUBLIC d3d11 d3d12 dxgi d3dcompiler windowsapp)
if(MSVC)
  target_compile_options(gpu_capture_bridge PRIVATE /W4 /EHsc /permissive-)
endif()
option(GPU_BRIDGE_BUILD_TESTS "Build headless hardware transport test" ON)
if(GPU_BRIDGE_BUILD_TESTS)
  enable_testing()
  add_executable(gpu_bridge_test synthetic_test.cpp)
  target_link_libraries(gpu_bridge_test PRIVATE gpu_capture_bridge)
  add_test(NAME gpu_bridge_synthetic COMMAND gpu_bridge_test)
  set_tests_properties(gpu_bridge_synthetic PROPERTIES TIMEOUT 90)
endif()
