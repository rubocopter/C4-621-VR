cmake_minimum_required(VERSION 3.20)
project(ac6_frame_pair_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../reference/minhook minhook)
enable_testing()
add_executable(test_pair_gate test_gate.cpp)
target_compile_definitions(test_pair_gate PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(test_pair_gate PRIVATE d3d12 dxgi user32)
add_test(NAME pair_failure_is_terminal COMMAND test_pair_gate)
add_executable(test_pair_ledger test_ledger.cpp)
target_compile_definitions(test_pair_ledger PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME pair_ledger_requires_same_recording COMMAND test_pair_ledger)
add_executable(test_frame_pair test_host.cpp probe.cpp slots.c)
target_compile_definitions(test_frame_pair PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX FRAME_PAIR_TEST_HOST)
target_link_libraries(test_frame_pair PRIVATE minhook d3d12 dxgi user32)
add_test(NAME frame_pair_matches_changing_gpu_fixture COMMAND test_frame_pair)
add_test(NAME frame_pair_cancels_when_resized COMMAND test_frame_pair --resize-test)
set_tests_properties(frame_pair_matches_changing_gpu_fixture PROPERTIES TIMEOUT 30)
add_library(ac6_frame_pair_probe SHARED probe.cpp slots.c)
set(FRAME_PAIR_PROBE_NAME "ac6_frame_pair_probe" CACHE STRING "Unique filename for a revised probe in an existing diagnostic session")
set_target_properties(ac6_frame_pair_probe PROPERTIES OUTPUT_NAME "${FRAME_PAIR_PROBE_NAME}")
target_compile_definitions(ac6_frame_pair_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ac6_frame_pair_probe PRIVATE minhook d3d12 dxgi user32)
