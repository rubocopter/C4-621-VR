cmake_minimum_required(VERSION 3.20)
project(ac6_projection_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../reference/minhook minhook)
add_executable(test_projection_snapshot test_snapshot.cpp)
target_compile_definitions(test_projection_snapshot PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
enable_testing()
add_test(NAME projection_snapshot_filters_and_preserves COMMAND test_projection_snapshot)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/probe.cpp")
  add_library(ac6_projection_probe SHARED probe.cpp)
  target_compile_definitions(ac6_projection_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(ac6_projection_probe PRIVATE minhook)
endif()
