cmake_minimum_required(VERSION 3.20)
project(ac6_native_constants_trace LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../reference/minhook minhook)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(trace_tests test_trace.cpp trace.cpp)
target_link_libraries(trace_tests PRIVATE minhook)
enable_testing()
add_test(NAME native_trace_preserves_original_and_bounds_reads COMMAND trace_tests)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/worker.cpp")
  add_library(ac6_render_probe SHARED worker.cpp trace.cpp)
  target_link_libraries(ac6_render_probe PRIVATE minhook user32)
endif()
