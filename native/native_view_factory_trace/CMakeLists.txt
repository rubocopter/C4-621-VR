cmake_minimum_required(VERSION 3.20)
project(ac6_native_view_factory_trace LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Requires MSVC x64 (SEH forwarding)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../reference/minhook minhook)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(factory_trace_core STATIC trace.cpp report.cpp)
target_compile_options(factory_trace_core PRIVATE /W4 /EHsc)
add_executable(factory_trace_tests test_trace.cpp)
target_compile_options(factory_trace_tests PRIVATE /W4 /EHsc /Ob0)
target_link_libraries(factory_trace_tests PRIVATE factory_trace_core minhook)
enable_testing()
foreach(case forwarding exceptions seh partial guard once concurrent minhook pending json)
  add_test(NAME native_view_factory_${case} COMMAND factory_trace_tests ${case})
endforeach()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/worker.cpp")
  add_library(ac6_render_probe SHARED worker.cpp)
  target_compile_options(ac6_render_probe PRIVATE /W4 /EHsc)
  target_link_libraries(ac6_render_probe PRIVATE factory_trace_core minhook user32)
endif()
