cmake_minimum_required(VERSION 3.20)
project(ac6_offline_stereo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/reprojection.hlsl")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/reprojection.hlsl" REPROJECTION_HLSL)
configure_file(shader.hpp.in shader.hpp @ONLY)
add_executable(ac6_offline_stereo main.cpp)
target_include_directories(ac6_offline_stereo PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(ac6_offline_stereo PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ac6_offline_stereo PRIVATE d3d12 dxgi d3dcompiler)
enable_testing()
add_test(NAME stereo_gpu_matches_fixture_and_nearest_surface COMMAND ac6_offline_stereo --self-test)
set_tests_properties(stereo_gpu_matches_fixture_and_nearest_surface PROPERTIES TIMEOUT 40)
