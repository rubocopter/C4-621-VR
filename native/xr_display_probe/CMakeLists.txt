cmake_minimum_required(VERSION 3.20)
project(ac6_xr_display_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(test_frame test_frame.cpp)
enable_testing()
add_test(NAME window_frame_preserves_channels_stride_and_aspect COMMAND test_frame)
add_executable(test_head_rotation test_head_rotation.cpp)
add_test(NAME head_rotation_recenters_clamps_and_uses_local_axes COMMAND test_head_rotation)
add_executable(test_static_stereo test_static_stereo.cpp)
add_test(NAME static_stereo_preserves_eye_order_and_rejects_bad_size COMMAND test_static_stereo)
add_executable(test_stereo_frame test_stereo_frame.cpp)
add_test(NAME stereo_frame_expiry_releases_both_eyes_and_ends_frame COMMAND test_stereo_frame)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/display_probe.cpp")
  add_executable(ac6_xr_display_probe display_probe.cpp window_capture.cpp)
  target_include_directories(ac6_xr_display_probe PRIVATE ../../reference/openxr)
  target_compile_definitions(ac6_xr_display_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D12 XR_NO_PROTOTYPES)
  target_link_libraries(ac6_xr_display_probe PRIVATE d3d12 dxgi d3d11 windowsapp user32)
  set(GPU_BRIDGE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(../gpu_capture_bridge gpu_capture_bridge)
  target_link_libraries(ac6_xr_display_probe PRIVATE gpu_capture_bridge)
  set(LIVE_STEREO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory(../live_stereo live_stereo)
  target_link_libraries(ac6_xr_display_probe PRIVATE live_stereo)
endif()
